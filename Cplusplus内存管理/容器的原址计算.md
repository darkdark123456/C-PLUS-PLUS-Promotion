// 使用c++原址操作字符串
// 使用类似的思想可以操作所有带有迭代器的容器
void transform_func() {

	//1 将vector中的所有元素缩放2.1倍
	//2 将字符串中的所有的字符转换为小写

	float scale = 2.1f;
	std::vector<float> vec_f = {1.0f,2.0f,3.0f};

	std::transform(vec_f.begin(), vec_f.end(), vec_f.begin(), [scale](float value)->float {
			
		return value * scale;
		});

	for (std::vector<float>::iterator iter = vec_f.begin(); iter != vec_f.end(); ++iter) {
		std::cout << *iter << "\n";
	}

	std::string str = "MENGZHENCHUAN";
	std::transform(str.begin(), str.end(), str.begin(), [](char c) -> char {
		
		return std::tolower(c);
		});
}

template <class _InIt, class _OutIt, class _Fn>
_CONSTEXPR20 _OutIt transform(const _InIt _First, const _InIt _Last, _OutIt _Dest, _Fn _Func) {
    // transform [_First, _Last) with _Func
    _Adl_verify_range(_First, _Last);
    auto _UFirst      = _Get_unwrapped(_First);
    const auto _ULast = _Get_unwrapped(_Last);
    auto _UDest       = _Get_unwrapped_n(_Dest, _Idl_distance<_InIt>(_UFirst, _ULast));
    for (; _UFirst != _ULast; ++_UFirst, (void) ++_UDest) {
        *_UDest = _Func(*_UFirst);
    }

    _Seek_wrapped(_Dest, _UDest);
    return _Dest;
}
